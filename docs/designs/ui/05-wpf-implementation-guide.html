4.2 Navigation (MVVM)

Use CommunityToolkit.Mvvm for [ObservableProperty], [RelayCommand]. Sidebar navigation items are RadioButton styled as nav items, grouped so only one is active. Bind IsChecked to VM property and switch CurrentPage.

4.3 Wizard Navigation

// WizardViewModel.cs
public partial class WizardViewModel : ObservableObject
{
    [ObservableProperty] private int currentStep = 0;
    [ObservableProperty] private ObservableObject currentPage;

    private readonly List<ObservableObject> _pages = new()
    {
        new WelcomeViewModel(),
        new ConnectSqlViewModel(),
        new DownstreamAuthViewModel(),
        // ... 9 pages total
    };

    [RelayCommand]
    private void Next() { if (CurrentStep < 8) CurrentPage = _pages[++CurrentStep]; }

    [RelayCommand]
    private void Back() { if (CurrentStep > 0) CurrentPage = _pages[--CurrentStep]; }
}

6. IPC Integration (Named Pipes + StreamJsonRpc)

// IpcClient.cs — connects to Windows Service
public class IpcClient : IAsyncDisposable
{
    private NamedPipeClientStream _pipe;
    private JsonRpc _rpc;

    public async Task ConnectAsync(string pipeName = "SQLExtractorIPC")
    {
        _pipe = new NamedPipeClientStream(".", pipeName, PipeDirection.InOut, PipeOptions.Asynchronous);
        await _pipe.ConnectAsync(timeout: 5000);
        _rpc = JsonRpc.Attach(_pipe);

        // Subscribe to log stream
        _rpc.AddLocalRpcMethod("onLogEntry", new Action<LogEntry>(OnLogReceived));
    }

    public async Task<ServiceStatus> GetStatusAsync()
        => await _rpc.InvokeAsync<ServiceStatus>("getStatus");

    public async Task TriggerRunAsync()
        => await _rpc.InvokeAsync("triggerRun");

    public async Task SubscribeLogsAsync()
        => await _rpc.InvokeAsync("subscribeLogs");

    public event Action<LogEntry> LogReceived;
    private void OnLogReceived(LogEntry entry)
        => LogReceived?.Invoke(entry);
}

8. Accessibility Notes

AspectGuideline
Color contrastAll text meets WCAG AA (4.5:1 for normal text). Primary on white = 4.6:1 OK.
Focus indicatorsUse focus ring (3px blue glow) on all interactive controls
Screen readerSet AutomationProperties.Name on all buttons, tags, status indicators
Keyboard navTab order through all controls. Escape closes dialogs. Enter activates primary button.
High contrastTest with Windows High Contrast mode. Use SystemColors fallback.
